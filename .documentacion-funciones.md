<!-- URL base de la API del Met: la parte que siempre es igual, sea lo que sea que estemos pidiendo (búsqueda o detalle de una obra) -->
const URL_BASE = 'https://collectionapi.metmuseum.org/public/collection/v1';

<!-- // Función genérica para pedir datos a la API del Met. "accion" es la parte de la URL que cambia según lo que necesitemos:"search?q=painting&hasImages=true&isPublicDomain=true" para buscar obras "objects/12345" para obtener el detalle de una obra concreta -->
const peticionMet = async (accion) => {
  try {
    <!-- Hacemos la petición a la URL completa (base + acción) y esperamos (await) a que el servidor responda -->
    const peticion = await fetch(`${URL_BASE}/${accion}`);

<!-- La respuesta llega en formato JSON "en bruto". .json() la convierte en un objeto de JavaScript que ya podemos usar (esto también tarda, por eso otro await) -->
    const datos = await peticion.json();

<!-- Devolvemos los datos ya listos para usar -->
    return datos;
  } catch (error) {
<!--  Si algo falla (sin conexión, error del servidor, etc.), lo mostramos por consola en vez de romper la aplicación -->
    console.log(error);
  }
};

<!-- Obtiene una obra aleatoria de la colección del Met.Hace dos peticiones: una para buscar IDs de obras que cumplan los filtros, y otra para obtener el detalle completo de una obra elegida al azar. -->
const getRandomArtwork = async () => {
  try {
    <!-- 1. Buscamos obras que sean "painting" y tengan imagen disponible.La API nos devuelve un objeto con "total" (nº de resultados) y "objectIDs" (array con los IDs de esas obras) -->
    const busqueda = await peticionMet("search?q=painting&hasImages=true");

  <!-- Extraemos solo el array de IDs que nos interesa -->
    const { objectIDs } = busqueda;

 <!--  2. Elegimos un índice al azar dentro del array de IDs (un número entero entre 0 y objectIDs.length - 1) -->
    const indiceAleatorio = Math.floor(Math.random() * objectIDs.length);

  <!--   Obtenemos el ID concreto de la obra que nos ha tocado -->
    const idObra = objectIDs[indiceAleatorio];

<!-- 3. Pedimos el detalle completo de esa obra usando su ID (título, artista, departamento, imagen, etc.) -->
    const detalleObra = await peticionMet(`objects/${idObra}`);

<!-- 4. Devolvemos los datos de la obra para que quien llame a esta función pueda mostrarlos -->
    return detalleObra;
  } catch (error) {
<!-- Si algo falla en cualquiera de las peticiones, lo mostramos por consola -->
    console.log(error);
  }
};

<!-- Genera las 4 opciones para la pregunta del juego (1 correcta + 3 señuelo).
Recibe la obra correcta y el campo sobre el que se pregunta ("title" o "artistDisplayName"), y devuelve un array de 4 valores desordenados, listo para mostrar en un <select>. -->
export const generarOpciones = async (obraCorrecta, campo) => {
  try {
<!--  Extraemos el valor correcto (título o artista) de la obra correcta,usando "campo" como nombre de propiedad dinámico -->
    const valorCorrecto = obraCorrecta[campo];

<!-- Pedimos 3 obras aleatorias más, que servirán como "señuelos" (opciones incorrectas) -->
    const senuelo1 = await getRandomArtwork();
    const senuelo2 = await getRandomArtwork();
    const senuelo3 = await getRandomArtwork();

<!-- De cada señuelo, extraemos el mismo campo (title o artistDisplayName)
    que estamos usando para la pregunta -->
    const valorSenuelo1 = senuelo1[campo];
    const valorSenuelo2 = senuelo2[campo];
    const valorSenuelo3 = senuelo3[campo];

<!-- Juntamos el valor correcto y los 3 señuelos en un array, y lo desordenamos al azar con sort() + Math.random() (así la opción correcta no está siempre en la misma posición) -->
    return [valorCorrecto, valorSenuelo1, valorSenuelo2, valorSenuelo3].sort(() => Math.random() - 0.5);
  } catch (error) {
    <!-- Si alguna petición falla, lo mostramos por consola -->
    
    console.log(error);
  }
};
